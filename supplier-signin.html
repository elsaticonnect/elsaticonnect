/* ========================================================= ELSATI CONNECT FIX SUPPLIER PROFILE CREATION + SIGN-IN ========================================================= */ /* --------------------------------------------------------- 1. MAKE SURE SUPPLIER PROFILE COLUMNS EXIST --------------------------------------------------------- */ alter table public.profiles add column if not exists role text; alter table public.profiles add column if not exists company_name text; alter table public.profiles add column if not exists contact_person text; alter table public.profiles add column if not exists verification_status text; alter table public.profiles add column if not exists verified boolean; /* --------------------------------------------------------- 2. ALLOW A LOGGED-IN USER TO READ THEIR OWN PROFILE --------------------------------------------------------- */ alter table public.profiles enable row level security; drop policy if exists "Users can read own profile" on public.profiles; create policy "Users can read own profile" on public.profiles for select to authenticated using ( auth.uid() = id ); /* --------------------------------------------------------- 3. FUNCTION TO AUTOMATICALLY CREATE SUPPLIER PROFILE --------------------------------------------------------- */ create or replace function public.handle_new_supplier() returns trigger language plpgsql security definer set search_path = public as $$ begin insert into public.profiles ( id, role, company_name, contact_person, verification_status, verified ) values ( new.id, coalesce( new.raw_user_meta_data->>'role', 'supplier' ), new.raw_user_meta_data->>'company_name', new.raw_user_meta_data->>'contact_person', 'pending', false ) on conflict (id) do update set role = coalesce( excluded.role, public.profiles.role ), company_name = coalesce( excluded.company_name, public.profiles.company_name ), contact_person = coalesce( excluded.contact_person, public.profiles.contact_person ); return new; end; $$; /* --------------------------------------------------------- 4. REMOVE OLD VERSION OF TRIGGER IF IT EXISTS --------------------------------------------------------- */ drop trigger if exists "on_auth_user_created_supplier" on auth.users; /* --------------------------------------------------------- 5. CREATE NEW TRIGGER --------------------------------------------------------- */ create trigger "on_auth_user_created_supplier" after insert on auth.users for each row execute function public.handle_new_supplier(); /* --------------------------------------------------------- 6. FIX SUPPLIER ACCOUNTS THAT WERE ALREADY CREATED --------------------------------------------------------- */ insert into public.profiles ( id, role, company_name, contact_person, verification_status, verified ) select u.id, 'supplier', u.raw_user_meta_data->>'company_name', u.raw_user_meta_data->>'contact_person', 'pending', false from auth.users u where u.raw_user_meta_data->>'role' = 'supplier' and not exists ( select 1 from public.profiles p where p.id = u.id ); /* --------------------------------------------------------- 7. FIX EXISTING SUPPLIER PROFILES --------------------------------------------------------- */ update public.profiles p set role = 'supplier', company_name = coalesce( p.company_name, u.raw_user_meta_data->>'company_name' ), contact_person = coalesce( p.contact_person, u.raw_user_meta_data->>'contact_person' ), verification_status = coalesce( p.verification_status, 'pending' ), verified = coalesce( p.verified, false ) from auth.users u where p.id = u.id and u.raw_user_meta_data->>'role' = 'supplier'; /* --------------------------------------------------------- 8. CONFIRM SETUP --------------------------------------------------------- */ select 'Supplier profile system is ready.' as result;
